CREATE TABLE IF NOT EXISTS BL_DM.DIM_DATE (
    transaction_date DATE PRIMARY KEY,
    day              SMALLINT,
    month            SMALLINT,
    month_name       VARCHAR(10),
    quarter          SMALLINT,
    year             SMALLINT,
    weekday          SMALLINT
);

INSERT INTO BL_DM.DIM_DATE (
    transaction_date,
    day,
    month,
    month_name,
    quarter,
    year,
    weekday
)
SELECT
    d::DATE                            AS transaction_date,
    EXTRACT(DAY FROM d)                AS day,
    EXTRACT(MONTH FROM d)              AS month,
    TO_CHAR(d, 'Month')                AS month_name,
    EXTRACT(QUARTER FROM d)            AS quarter,
    EXTRACT(YEAR FROM d)               AS year,
    EXTRACT(DOW FROM d)                AS weekday
FROM generate_series(
    DATE '1980-01-01',
    DATE '2035-12-31',
    INTERVAL '1 day'
) AS d;

CREATE TABLE IF NOT EXISTS bl_dm.dim_customer (
customer_surr_id int PRIMARY KEY,
customer_gender varchar NOT NULL,
customer_birthday date NOT NULL,
source_id int NOT NULL,
source_system varchar NOT NULL,
source_entity varchar NOT NULL,
insert_dt date NOT NULL,
update_dt date NOT NULL,
CONSTRAINT customer_bday_fk FOREIGN KEY (customer_birthday) REFERENCES bl_dm.dim_date (transaction_date),
CONSTRAINT customer_source_fk FOREIGN KEY (source_id) REFERENCES bl_dm.dim_source (source_id)
);

CREATE TABLE IF NOT EXISTS  bl_dm.dim_cashier (
cashier_surr_id int PRIMARY KEY,

source_system varchar NOT NULL,

insert_dt date NOT NULL,
start_dt date NOT NULL,
end_dt date NOT NULL,
is_active boolean NOT NULL,
CONSTRAINT cashier_bday_fk FOREIGN KEY (cashier_birthday) REFERENCES bl_dm.dim_date (transaction_date),
CONSTRAINT cashier_source_fk FOREIGN KEY (source_id) REFERENCES bl_dm.dim_source (source_id)
);

CREATE TABLE IF NOT EXISTS fact_transactions (
transaction_surr_id INT PRIMARY KEY,
transaction_date DATE NOT NULL,
store_id INT NOT NULL,
cashier_surr_id INT NOT NULL,
customer_surr_id INT NOT NULL,
payment_method_id INT NOT NULL,
items_count INT NOT NULL,
total_amount_ngn DECIMAL NOT NULL,
avg_price DECIMAL GENERATED ALWAYS AS (total_amount_ngn / items_count) STORED,
source_id int NOT NULL,
source_system varchar NOT NULL,
source_entity varchar NOT NULL,
insert_dt date NOT NULL,
update_dt date NOT NULL,
CONSTRAINT transaction_store_fk FOREIGN KEY (store_id) REFERENCES bl_dm.dim_store (store_id),
CONSTRAINT transaction_cashier_fk FOREIGN KEY (cashier_surr_id) REFERENCES bl_dm.dim_cashier (cashier_surr_id),
CONSTRAINT transaction_customer_fk FOREIGN KEY (customer_surr_id) REFERENCES bl_dm.dim_customer (customer_surr_id),
CONSTRAINT transaction_pm_fk FOREIGN KEY (payment_method_id) REFERENCES bl_dm.dim_payment_method (payment_method_id),
CONSTRAINT transaction_source_fk FOREIGN KEY (source_id) REFERENCES bl_dm.dim_source (source_id)
);
