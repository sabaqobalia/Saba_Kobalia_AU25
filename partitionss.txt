--create procedure for inseting from source 1 and 2.
CREATE OR REPLACE PROCEDURE bl_cl.load_ce_transactions_src1 ()
LANGUAGE plpgsql
AS $$

DECLARE
	v_source int;
	v_rows int;
BEGIN
	SELECT source_id INTO v_source
	FROM bl_3nf.ce_source 
	WHERE source_entity = 'src_source1';


	INSERT INTO bl_3nf.ce_transactions (transaction_src_id,store_id,
	transaction_date,cashier_surr_id,customer_surr_id,items_count,total_amount_ngn,discount_applied,
	loyalty_points_earned,payment_method_id,source_id,source_system,source_entity,insert_dt,update_dt)
	SELECT
	transaction_id,st.store_id,
	transaction_date::date,cs.cashier_surr_id,cu.customer_surr_id,
	items_count::smallint
	,total_amount_ngn::numeric,discount_applied::boolean
	,loyalty_points_earned::int,
	pm.payment_method_id,
	v_source, 'sa_source1', 'src_source1', current_date,current_date
	FROM sa_source1.src_source1 s 
	 JOIN bl_3nf.ce_store st 
	ON s.store_name = st.store_name  
	 JOIN bl_3nf.ce_city ci 
	ON	ci.city_id = st.city_id AND ci.city_name = s.city
	 JOIN bl_3nf.ce_customer cu 
	ON cu.customer_src_id = s.customer_id
	 JOIN bl_3nf.ce_cashier cs
	ON cs.cashier_src_id = s.cashier_id
	 JOIN bl_3nf.ce_payment_method pm
	ON pm.payment_method_name = s.payment_method

	ON CONFLICT (transaction_src_id,source_id,transaction_date) DO NOTHING;
GET DIAGNOSTICS v_rows = ROW_COUNT;

 CALL bl_cl.log_event(
        'load_ce_transactions_src1',
        v_rows,
        'Transactions source1 loaded'
    );

EXCEPTION
    WHEN OTHERS THEN
        CALL bl_cl.log_event(
            'load_ce_transactions_src1',
            0,
            SQLERRM
        );
END;
$$;

CREATE OR REPLACE PROCEDURE load_fact_transactions ()
LANGUAGE plpgsql
AS $$
DECLARE
	v_rows int :=0;
	rec RECORD;
	CUR REFCURSOR;
	v_source int;

BEGIN
	SELECT source_id INTO v_source
	FROM bl_dm.dim_source
	WHERE source_entity = 'ce_transactions';

	OPEN cur FOR
		SELECT transaction_surr_id,transaction_date,store_id,cashier_surr_id,
			   customer_surr_id,payment_method_id,items_count,total_amount_ngn
		FROM bl_3nf.ce_transactions;

	LOOP
		   FETCH cur INTO rec;
		   EXIT WHEN NOT FOUND;
	
		   INSERT INTO bl_dm.fact_transactions(
		   transaction_surr_id,transaction_date,store_id,cashier_surr_id,customer_surr_id,
		   payment_method_id,items_count,total_amount_ngn,source_id,source_system,
		   source_entity,insert_dt,update_dt
		   )
		   VALUES
		   (rec.transaction_surr_id,rec.transaction_date,rec.store_id,rec.cashier_surr_id,
		   rec.customer_surr_id,rec.payment_method_id,rec.items_count,rec.total_amount_ngn,
		   v_source,'bl_3nf','ce_transactions',current_date,current_date
		   )
		   ON CONFLICT (transaction_surr_id,transaction_date) DO NOTHING;
	
		   IF FOUND THEN v_rows := v_rows +1;
		   END IF;
	END LOOP;
	CLOSE cur;
 CALL bl_cl.log_event('load_fact_transactions', v_rows, 'Fact transactions load successful');
    
EXCEPTION
    WHEN OTHERS THEN
        CALL bl_cl.log_event('load_fact_transactions', 0, SQLERRM);
        RAISE;
END
$$;
