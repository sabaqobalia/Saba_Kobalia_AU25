--function and procedure for logging table
CREATE TABLE bl_cl.etl_log (
    log_id int GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    log_timestamp timestamp DEFAULT now(),
    procedure_name varchar(100),
    rows_affected int,
    log_message text
);

CREATE OR REPLACE PROCEDURE bl_cl.log_event(
    p_procedure_name varchar,
    p_rows int,
    p_message text
)
LANGUAGE plpgsql
AS $$
BEGIN
    INSERT INTO bl_cl.etl_log(procedure_name, rows_affected, log_message)
    VALUES (p_procedure_name, p_rows, p_message);
END;
$$;

--function and procedure for ce_city:

CREATE OR REPLACE FUNCTION bl_cl.get_city_data()
RETURNS TABLE(
    city_id int,
    city_name varchar
)
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    SELECT
        mc.city_id,
        mc.city_name
    FROM bl_cl.map_city mc;
END;
$$;


CREATE OR REPLACE PROCEDURE bl_cl.load_ce_city()
LANGUAGE plpgsql
AS $$
DECLARE
    rec RECORD;
    v_rows int := 0;
    v_source_id int;
BEGIN

    SELECT source_id
    INTO v_source_id
    FROM bl_3nf.ce_source
    WHERE source_entity = 'map_city';

    FOR rec IN SELECT * FROM bl_cl.get_city_data()
    LOOP
        INSERT INTO bl_3nf.ce_city (
            city_id,
            city_name,
            source_id,
            source_entity,
            source_system,
            insert_dt,
            update_dt
        )
        VALUES (
            rec.city_id,
            rec.city_name,
            v_source_id,
            'map_city',
            'bl_cl',
            current_date,
            current_date
        )
        ON CONFLICT (city_id) DO NOTHING;

        v_rows := v_rows + 1;
    END LOOP;

    CALL bl_cl.log_event(
        'load_ce_city',
        v_rows,
        'City load successful'
    );

EXCEPTION
    WHEN OTHERS THEN
        CALL bl_cl.log_event(
            'load_ce_city',
            0,
            SQLERRM
        );
END;
$$;

--two procedures to load transactions from source1 and source2:
CREATE OR REPLACE PROCEDURE bl_cl.load_ce_transactions_src1()
LANGUAGE plpgsql
AS $$
DECLARE
    v_rows int;
BEGIN

    INSERT INTO bl_3nf.ce_transactions (
        transaction_src_id,
        store_id,
        transaction_date,
        cashier_surr_id,
        customer_surr_id,
        items_count,
        total_amount_ngn,
        discount_applied,
        loyalty_points_earned,
        payment_method_id,
        source_id,
        source_system,
        source_entity,
        insert_dt,
        update_dt
    )
    SELECT
        src.transaction_id,
        s.store_id,
        src.transaction_date::date,
        c.cashier_surr_id,
        cu.customer_surr_id,
        src.items_count::smallint,
        src.total_amount_ngn::numeric,
        src.discount_applied::boolean,
        src.loyalty_points_earned::int,
        pm.payment_method_id,
        so.source_id,
        'sa_source1',
        'src_source1',
        current_date,
        current_date
    FROM sa_source1.src_source1 src
    JOIN bl_3nf.ce_store s
        ON s.store_name = src.store_name
    JOIN bl_3nf.ce_city ci
        ON ci.city_id = s.city_id
       AND ci.city_name = src.city
    JOIN bl_3nf.ce_cashier c
        ON c.cashier_src_id = src.cashier_id
    JOIN bl_3nf.ce_customer cu
        ON cu.customer_src_id = src.customer_id
    JOIN bl_3nf.ce_payment_method pm
        ON pm.payment_method_name = src.payment_method
    JOIN bl_3nf.ce_source so
        ON so.source_entity = 'src_source1'
    ON CONFLICT (transaction_src_id, source_id) DO NOTHING;

    GET DIAGNOSTICS v_rows = ROW_COUNT;

    CALL bl_cl.log_event(
        'load_ce_transactions_src1',
        v_rows,
        'Transactions source1 loaded'
    );

EXCEPTION
    WHEN OTHERS THEN
        CALL bl_cl.log_event(
            'load_ce_transactions_src1',
            0,
            SQLERRM
        );
END;
$$;

--procedures to link cashiers and stores:
CREATE OR REPLACE PROCEDURE bl_cl.load_ce_store_cashier()
LANGUAGE plpgsql
AS $$
DECLARE
    v_rows int := 0;
    v_source_id int;
BEGIN

    SELECT source_id
    INTO v_source_id
    FROM bl_3nf.ce_source
    WHERE source_entity = 'ce_transactions';

    INSERT INTO bl_3nf.ce_store_cashier (
        store_id,
        cashier_surr_id,
        source_system,
        source_id,
        source_entity,
        start_dt,
        end_dt,
        insert_dt,
        is_active
    )
    SELECT 
        t.store_id,
        t.cashier_surr_id,
        'bl_3nf',
        v_source_id,
        'ce_transactions',
        current_date,
        DATE '9999-12-31',
        current_date,
        TRUE
    FROM (
        SELECT
            cashier_surr_id,
            store_id,
            ROW_NUMBER() OVER (
                PARTITION BY cashier_surr_id
                ORDER BY transaction_date DESC
            ) AS rn
        FROM bl_3nf.ce_transactions
    ) t
    WHERE t.rn = 1
    ON CONFLICT (store_id,cashier_surr_id,start_dt) DO NOTHING;

    GET DIAGNOSTICS v_rows = ROW_COUNT;

    CALL bl_cl.log_event(
        'load_ce_store_cashier',
        v_rows,
        'Store-Cashier bridge loaded'
    );

EXCEPTION
    WHEN OTHERS THEN
        CALL bl_cl.log_event(
            'load_ce_store_cashier',
            0,
            SQLERRM
        );
END;
$$;
